    55→// fetch — native http (Wasm yok)
    56→globalThis.fetch = function(url, options) {
    57→  options = options || {};
    58→  return new Promise(function(resolve, reject) {
    59→    var lib = url.indexOf('https') === 0 ? require('https') : require('http');
    60→    var urlObj = new URL(url);
    61→    var headers = Object.assign({}, options.headers || {});
    62→    if (options.body) headers['Content-Length'] = Buffer.byteLength(options.body);
    63→    var req = lib.request({
    64→      hostname: urlObj.hostname,
    65→      port: urlObj.port || (urlObj.protocol === 'https:' ? 443 : 80),
    66→      path: urlObj.pathname + urlObj.search,
    67→      method: options.method || 'GET',
    68→      headers: headers
    69→    }, function(resp) {
    70→      var chunks = [];
    71→      resp.on('data', function(c) { chunks.push(c); });
    72→      resp.on('end', function() {
    73→        var text = Buffer.concat(chunks).toString('utf8');
    74→        resolve({ status: resp.statusCode, ok: resp.statusCode >= 200 && resp.statusCode < 300, json: function() { return Promise.resolve(JSON.parse(text)); }, text: function() { return Promise.resolve(text); } });
    75→      });
    76→    });
    77→    req.on('error', reject);
    78→    req.setTimeout(120000, function() { req.destroy(); reject(new Error('timeout')); });
    79→    if (options.body) req.write(options.body);
    80→    req.end();
    81→  });
    82→};

   262→  // Plagiarizm kontrolü — AI cevabında 4+ kelimelik ardışık dizilim kaynak metinde var mı?
   263→  function findPlagiarism(aiText, sourceText) {
   264→    var normalize = function(t) {
   265→      return String(t || '')
   266→        .toLowerCase()
   267→        .replace(/[''`]/g, "'")
   268→        .replace(/[İI]/g, 'i')
   269→        .replace(/Ş/g, 's').replace(/Ç/g, 'c').replace(/Ğ/g, 'g').replace(/Ü/g, 'u').replace(/Ö/g, 'o')
   270→        .replace(/ı/g, 'i').replace(/ş/g, 's').replace(/ç/g, 'c').replace(/ğ/g, 'g').replace(/ü/g, 'u').replace(/ö/g, 'o')
   271→        .replace(/[^\w\s]/g, ' ')
   272→        .replace(/\s+/g, ' ')
   273→        .trim();
   274→    };
   275→    var ai = normalize(aiText);
   276→    var src = normalize(sourceText);
   277→    if (!ai || !src) return [];
   278→
   279→    var aiWords = ai.split(' ');
   280→    var found = [];
   281→    var seen = new Set();
   282→    // 4 kelimelik ardışık dizilimleri kaynak metinde ara
   283→    for (var i = 0; i + 4 <= aiWords.length; i++) {
   284→      var chunk = aiWords.slice(i, i + 4).join(' ');
   285→      // 15+ karakter ve sadece jenerik olmayan (en az 1 meaningful kelime içersin)
   286→      if (chunk.length > 15 && src.indexOf(chunk) >= 0 && !seen.has(chunk)) {
   287→        seen.add(chunk);
   288→        found.push(chunk);
   289→      }
   290→    }
   291→    return found;
   292→  }

   322→        deadKeys.add(currentKey);
   323→        log('  AI 429 kota dolu — key ölü (bu cycle atlanacak)');

   325→      }
   326→      if (result.error && result.error.code === 503) {
   327→        log('  AI 503 — 5 sn bekle');
   328→        await new Promise(function(r) { setTimeout(r, 5000); });
   329→        attempt--; keyIndex--;
   330→        continue;
   331→      }
   332→      if (result.error) {
   333→        log('  AI error ' + result.error.code + ': ' + (result.error.message || '').slice(0, 80));
   334→        deadKeys.add(currentKey);
   335→        continue;
   336→      }
   337→      deadKeys.delete(currentKey);
   338→      if (result.candidates && result.candidates.length > 0 && result.candidates[0].content && result.candidates[0].content.parts && result.candidates[0].content.parts.length > 0) {
   339→        var text = result.candidates[0].content.parts[0].text;
   340→        if (text) {
   341→          text = text.trim();
   342→          var wc = countWords(text);
   343→
   344→          // Plagiarizm kontrolü — kaynak metinle 4+ kelimelik ardışık dizilim ara
   345→          var plagiarism = findPlagiarism(text, combinedContent);
   346→
   347→          if (plagiarism.length > 0) {
   348→            log('  AI özet: ' + wc + ' kelime — ' + plagiarism.length + ' kopyalama tespit edildi (örn: "' + plagiarism[0].slice(0, 50) + '")');
   349→            // bestText olarak kabul et ama retry için geri bildirim ver
   350→            if (wc > bestWordCount) { bestText = text; bestWordCount = wc; }
   351→            prevAttemptText = text;
   352→            prevPlagiarism = plagiarism;
   353→            prevWordCount = wc;
   354→            prevAds = null;  // reklam temizle, sadece plagiarizm retry
   355→            // Yeniden deneme yapılacak (return etme)
   356→            continue;
   357→          }
   358→
   359→          // Reklam tespiti — AI cevabında reklam/CTA ifadeleri var mı?
   360→          var ads = findAds(text);
   361→          if (ads.length > 0) {
   362→            log('  AI özet: ' + wc + ' kelime — ' + ads.length + ' reklam tespit edildi (örn: "' + ads[0] + '")');
   363→            if (wc > bestWordCount) { bestText = text; bestWordCount = wc; }
   364→            prevAttemptText = text;
   365→            prevPlagiarism = null;  // plagiarizm temizle, sadece reklam retry
   366→            prevWordCount = wc;
   367→            prevAds = ads;
   368→            continue;
   369→          }
   370→
   371→          // Yetersiz kelime kontrolü — min'den azsa retry yap (AI'a "daha uzun yaz" geri bildirimi)
   372→          if (wc < minWords) {
   373→            log('  AI özet: ' + wc + ' kelime (min: ' + minWords + ') — YETERSİZ, tekrar denenecek');
   374→            if (wc > bestWordCount) { bestText = text; bestWordCount = wc; }
   375→            prevAttemptText = text;
   376→            prevPlagiarism = null;  // plagiarizm temizle, sadece kelime retry
   377→            prevWordCount = wc;
   378→            continue;
   379→          }
   380→
   381→          log('  AI özet: ' + wc + ' kelime (min: ' + minWords + ') — kopyalama YOK, kelime YETERLİ');
   382→          if (wc > bestWordCount) { bestText = text; bestWordCount = wc; }
   383→          if (wc >= minWords) return text;
   384→          continue;
   385→        }
   386→      }
   387→      continue;
   388→    } catch (e) {
   389→      log('  AI hata: ' + e.message);
   390→      deadKeys.add(currentKey);
   391→      continue;
   392→    }
   393→  }
   394→  return bestText;
   395→}

   397→// Başlık benzerliği — HTML entity decode + Siyasetçi isimleri tek başına yeterli değil, en az 2 anlamlı kelime
   398→function normalizeTitle(t) {
   399→  if (!t) return '';
   400→  // HTML entity decode (&#039; -> ', &amp; -> &, &quot; -> ", &nbsp; -> space, &#x27; -> ')
   401→  return String(t)
   402→    .replace(/&#x([0-9a-fA-F]+);/g, function(_, h) { return String.fromCharCode(parseInt(h, 16)); })
   403→    .replace(/&#(\d+);/g, function(_, d) { return String.fromCharCode(parseInt(d, 10)); })
   404→    .replace(/&amp;/g, '&')
   405→    .replace(/&quot;/g, '"')
   406→    .replace(/&apos;/g, "'")
   407→    .replace(/&nbsp;/g, ' ')
   408→    .replace(/&lt;/g, '<')
   409→    .replace(/&gt;/g, '>')
   410→    .toLowerCase()
   411→    .replace(/[''`]/g, "'")
   412→    // Ay ismi + ek → ay ismi (eylülde → eylül, kasımda → kasım, ekim'in → ekim)
   413→    .replace(/\b(ocak|şubat|subat|mart|nisan|mayıs|mayis|haziran|temmuz|ağustos|agustos|eylül|eylul|ekim|kasım|kasim|aralık|aralik)(de|da|te|ta|in|ın|un|ün|nin|nın|nun|nün|den|dan|ten|tan|e|a|ye|ya|yi|yı|yu|yü|nden|ndan|nde|nda)\b/g, '$1')
   414→    // Belediye ek'leri → belediye (belediyesinde → belediye)
   415→    .replace(/\b(belediye)(si|sinde|sine|sinin|sinde|sinden|siyle)\b/g, '$1')
   416→    // Soruşturma ek'leri → soruşturma
   417→    .replace(/\b(soruşturma|sorusturma)(sı|si|sında|sinda|sinden|sının|sinin)\b/g, '$1')
   418→    .replace(/[^\w\sçğıöşü]/g, ' ')
   419→    .replace(/\s+/g, ' ')
   420→    .trim();
   421→}

   477→function titleSimilar(t1, t2) {
   478→  var n1 = normalizeTitle(t1), n2 = normalizeTitle(t2);
   479→  if (!n1 || !n2) return 0;
   480→  // Birebir aynı başlık → kesin eşleşme (%100)
   481→  if (n1 === n2) return 1.0;
   482→  // 4+ karakter ve stop word olmayan kelimeleri al
   483→  var w1 = n1.split(' ').filter(function(w) { return w.length > 3 && !STOP_WORDS.has(w); });
   484→  var w2 = n2.split(' ').filter(function(w) { return w.length > 3 && !STOP_WORDS.has(w); });
   485→  if (!w1.length || !w2.length) return 0;
   486→  var set2 = new Set(w2); var common = 0;
   487→  w1.forEach(function(w) { if (set2.has(w)) common++; });
   488→  var ratio = common / Math.max(w1.length, w2.length);
   489→  // En az 2 anlamlı ortak kelime şart (jenerik başlıklar birleşmesin)
   490→  if (common < 2) return 0;
   491→  // Eşik yüksek — sadece anlamlı örtüşme varsa kabul
   492→  if (ratio < 0.5) return 0;
   493→  return ratio;
   494→}

   496→// İçerik benzerliği — başlık yeterli değilse içeriğe bak
   497→function contentSimilar(c1, c2) {
   498→  if (!c1 || !c2) return 0;
   499→  var n1 = normalizeTitle(c1.slice(0, 1500));
   500→  var n2 = normalizeTitle(c2.slice(0, 1500));
   501→  var w1 = n1.split(' ').filter(function(w) { return w.length > 4 && !STOP_WORDS.has(w); });
   502→  var w2 = n2.split(' ').filter(function(w) { return w.length > 4 && !STOP_WORDS.has(w); });
   503→  if (!w1.length || !w2.length) return 0;
   504→  var set2 = new Set(w2); var common = 0;
   505→  w1.forEach(function(w) { if (set2.has(w)) common++; });
   506→  var ratio = common / Math.max(w1.length, w2.length);
   507→  // İçerikte en az 5 anlamlı ortak kelime şart
   508→  if (common < 5) return 0;
   509→  if (ratio < 0.5) return 0;
   510→  return ratio;
   511→}

   513→// Gruplama — başlık benzerliği yüksek VEYA (başlık orta + içerik yüksek)
   514→// Eşik YÜKSEK — alakasız haberler aynı grupta birleşmesin
   515→// "İstanbul baskını" vs "İstanbul gastronomi" -> titleSimilar düşük, ayrı grup
   516→// "Ataşehir Belediyesi soruşturması" vs "Vahap Seçer soruşturması" -> STOP'lar yeter,
   517→// meaningful kelimeler farklı (ataşehir+yolsuzluk vs vahap+seçer), titleSimilar düşük, ayrı grup
   518→function groupArticles(articles) {
   519→  var groups = [], used = new Set();
   520→  for (var i = 0; i < articles.length; i++) {
   521→    if (used.has(i)) continue;
   522→    var group = { articles: [articles[i]], sourceIds: new Set([articles[i].sourceId]) };
   523→    used.add(i);
   524→    for (var j = i + 1; j < articles.length; j++) {
   525→      if (used.has(j)) continue;
   526→      // Kategori BAĞIMSIZ — aynı haber farklı kategori etiketiyle gelirse yine eşleşir
   527→      var titleSim = titleSimilar(articles[i].title, articles[j].title);
   528→      var contentSim = contentSimilar(articles[i].content || articles[i].description || '', articles[j].content || articles[j].description || '');
   529→      // Eşik: title >= 0.5 (yüksek) VEYA (title >= 0.3 + content >= 0.5)
   530→      // Bu, alakasız haberlerin aynı grupta olmasını kesin önler
   531→      if (titleSim >= 0.5 || (titleSim >= 0.3 && contentSim >= 0.5)) {
   532→        group.articles.push(articles[j]);
   533→        group.sourceIds.add(articles[j].sourceId);
   534→        used.add(j);
   535→      }
   536→    }
   537→    groups.push(group);
   538→  }
   539→  return groups;
   540→}

   810→          var sonBilgilerPrefix = 'Konu ile ilgili son bilgiler şu şekildedir: ';
   811→          if (summaryText.toLowerCase().indexOf(sonBilgilerPrefix.toLowerCase()) !== 0) {

   821→            data: {
   822→              aiTitle: decodeHtmlEntities(firstArticle.title),
   823→              aiSummary: summaryText.slice(0, 5000),
   824→              category: cat,
   825→              imageUrl: bestImage,
   826→              sourceArticleIds: JSON.stringify(groupArticlesList.map(function(a) { return a.id; })),

   828→              earliestPublishedAt: groupArticlesList[groupArticlesList.length - 1].publishedAt || publishTime,
   829→              latestPublishedAt: publishTime,
   830→              wordCount: summaryText.split(/\s+/).length,
   831→              initialHearts: Math.floor(Math.random() * (413 - 223 + 1)) + 223, // random 223-413

   834→              publishedAt: publishTime
   835→            }

   844→            category: cat,
   845→            status: 'published'
