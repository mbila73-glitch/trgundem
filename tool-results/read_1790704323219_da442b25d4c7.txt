     1→'use client';
     2→
     3→import { useCallback, useEffect, useState } from 'react';
     4→import {
     5→  Lock, Trash2, Mail, Clock, Loader2, CheckSquare, Square, CheckCheck,
     6→  AlertTriangle, RotateCcw, ArrowLeft, ExternalLink, Save, Globe, Star,
     7→  Newspaper, FileText, FolderTree, Edit3, X, Upload
     8→} from 'lucide-react';
     9→import { toast } from 'sonner';
    10→import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
    11→import { Input } from '@/components/ui/input';
    12→import { Textarea } from '@/components/ui/textarea';
    13→import { Button } from '@/components/ui/button';
    14→import { Badge } from '@/components/ui/badge';
    15→import { Card } from '@/components/ui/card';
    16→import { Skeleton } from '@/components/ui/skeleton';
    17→import { Checkbox } from '@/components/ui/checkbox';
    18→import { Label } from '@/components/ui/label';
    19→import {
    20→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue
    21→} from '@/components/ui/select';
    22→import {
    23→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    24→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle, AlertDialogTrigger
    25→} from '@/components/ui/alert-dialog';
    26→
    27→type Message = { id: string; name: string; email: string; subject: string; message: string; status: string; createdAt: string };
    28→type PubArticle = { id: string; aiTitle: string; aiSummary: string; imageUrl: string | null; category: string; wordCount: number; sourceCount: number; sourceArticleIds: string; publishedAt: string | null; latestPublishedAt: string; };
    29→type AdminTab = 'messages' | 'custom' | 'published';
    30→
    31→const CATEGORIES = ['Güncel', 'Kamu / Resmi', 'Ekonomi / Finans', 'Spor / Magazin', 'Bilim / Teknoloji', 'Kültür / Sanat'];
    32→
    33→export function AdminPanel({ open, onClose }: { open: boolean; onClose: () => void }) {
    34→  const [token, setToken] = useState<string | null>(null);
    35→  const [password, setPassword] = useState('');
    36→  const [loggingIn, setLoggingIn] = useState(false);
    37→  const [adminTab, setAdminTab] = useState<AdminTab>('messages');
    38→
    39→  // Messages state
    40→  const [messages, setMessages] = useState<Message[]>([]);
    41→  const [loadingMsgs, setLoadingMsgs] = useState(false);
    42→  const [deletingId, setDeletingId] = useState<string | null>(null);
    43→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
    44→  const [bulkDeleting, setBulkDeleting] = useState(false);
    45→
    46→  // Reset state
    47→  const [resetOpen, setResetOpen] = useState(false);
    48→  const [resetConfirm, setResetConfirm] = useState('');
    49→  const [resetting, setResetting] = useState(false);
    50→  const [resetResult, setResetResult] = useState<string | null>(null);
    51→
    52→  // Custom article state
    53→  const [fetchUrl, setFetchUrl] = useState('');
    54→  const [fetching, setFetching] = useState(false);
    55→  const [customTitle, setCustomTitle] = useState('');
    56→  const [customSummary, setCustomSummary] = useState('');
    57→  const [customImage, setCustomImage] = useState('');
    58→  const [customCategory] = useState('Özel');
    59→  const [fetchedImages, setFetchedImages] = useState<string[]>([]);
    60→  const [saving, setSaving] = useState(false);
    61→
    62→  // Published articles state
    63→  const [pubArticles, setPubArticles] = useState<PubArticle[]>([]);
    64→  const [loadingPub, setLoadingPub] = useState(false);
    65→  const [editingId, setEditingId] = useState<string | null>(null);
    66→  const [editTitle, setEditTitle] = useState('');
    67→  const [editSummary, setEditSummary] = useState('');
    68→  const [editImage, setEditImage] = useState('');
    69→  const [savingEdit, setSavingEdit] = useState(false);
    70→
    71→  useEffect(() => { const s = localStorage.getItem('admin_token'); if (s) setToken(s); }, []);
    72→
    73→  const handleLogin = async (e: React.FormEvent) => {
    74→    e.preventDefault();
    75→    if (!password.trim()) return;
    76→    setLoggingIn(true);
    77→    try {
    78→      const r = await fetch('/api/admin/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ password }) });
    79→      const json = (await r.json()) as { ok?: boolean; token?: string; error?: string };
    80→      if (!r.ok || !json.ok) throw new Error(json.error || 'Giriş başarısız');
    81→      localStorage.setItem('admin_token', json.token!);
    82→      setToken(json.token!); setPassword('');
    83→      toast.success('Giriş başarılı');
    84→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
    85→    finally { setLoggingIn(false); }
    86→  };
    87→
    88→  const loadMessages = useCallback(async () => {
    89→    if (!token) return;
    90→    setLoadingMsgs(true);
    91→    try {
    92→      const r = await fetch('/api/admin/messages', { headers: { Authorization: `Bearer ${token}` } });
    93→      if (r.status === 401) { localStorage.removeItem('admin_token'); setToken(null); toast.error('Oturum süresi doldu'); return; }
    94→      const json = (await r.json()) as { messages?: Message[] };
    95→      setMessages(json.messages ?? []); setSelectedIds(new Set());
    96→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Yükleme hatası'); }
    97→    finally { setLoadingMsgs(false); }
    98→  }, [token]);
    99→
   100→  const loadPublished = useCallback(async () => {
   101→    if (!token) return;
   102→    setLoadingPub(true);
   103→    try {
   104→      const r = await fetch('/api/admin/published', { headers: { Authorization: `Bearer ${token}` } });
   105→      const json = (await r.json()) as { articles?: PubArticle[] };
   106→      setPubArticles(json.articles ?? []);
   107→    } catch { toast.error('Haberler yüklenemedi'); }
   108→    finally { setLoadingPub(false); }
   109→  }, [token]);
   110→
   111→  useEffect(() => {
   112→    if (open && token) {
   113→      if (adminTab === 'messages') void loadMessages();
   114→      if (adminTab === 'published') void loadPublished();
   115→    }
   116→  }, [open, token, adminTab, loadMessages, loadPublished]);
   117→
   118→  const handleDelete = async (id: string) => {
   119→    setDeletingId(id);
   120→    try { const r = await fetch(`/api/admin/messages/${id}`, { method: 'DELETE', headers: { Authorization: `Bearer ${token}` } });
   121→      if (!r.ok) throw new Error('Silinemedi');
   122→      setMessages(a => a.filter(m => m.id !== id));
   123→      setSelectedIds(p => { const n = new Set(p); n.delete(id); return n; });
   124→      toast.success('Mesaj silindi');
   125→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
   126→    finally { setDeletingId(null); }
   127→  };
   128→
   129→  const toggleSelect = (id: string) => setSelectedIds(p => { const n = new Set(p); if (n.has(id)) n.delete(id); else n.add(id); return n; });
   130→  const selectAll = () => setSelectedIds(selectedIds.size === messages.length ? new Set() : new Set(messages.map(m => m.id)));
   131→
   132→  const deleteSelected = async () => {
   133→    setBulkDeleting(true);
   134→    try { await Promise.all(Array.from(selectedIds).map(id => fetch(`/api/admin/messages/${id}`, { method: 'DELETE', headers: { Authorization: `Bearer ${token}` } })));
   135→      setMessages(a => a.filter(m => !selectedIds.has(m.id))); toast.success(`${selectedIds.size} mesaj silindi`); setSelectedIds(new Set());
   136→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
   137→    finally { setBulkDeleting(false); }
   138→  };
   139→
   140→  const deleteAllMsgs = async () => {
   141→    setBulkDeleting(true);
   142→    try { await Promise.all(messages.map(m => fetch(`/api/admin/messages/${m.id}`, { method: 'DELETE', headers: { Authorization: `Bearer ${token}` } })));
   143→      setMessages([]); setSelectedIds(new Set()); toast.success('Tüm mesajlar silindi');
   144→    } catch { toast.error('Tümünü silme hatası'); }
   145→    finally { setBulkDeleting(false); }
   146→  };
   147→
   148→  // Custom article handlers
   149→  const handleFetch = async (e: React.FormEvent) => {
   150→    e.preventDefault();
   151→    if (!fetchUrl.trim()) return;
   152→    setFetching(true);
   153→    try {
   154→      const r = await fetch('/api/admin/custom-article', { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` }, body: JSON.stringify({ action: 'fetch', url: fetchUrl }) });
   155→      const json = (await r.json()) as { ok?: boolean; title?: string; description?: string; images?: string[]; error?: string };
   156→      if (!r.ok || !json.ok) throw new Error(json.error || 'Getirilemedi');
   157→      setCustomTitle(json.title || '');
   158→      setCustomSummary(json.description || '');
   159→      setFetchedImages(json.images ?? []);
   160→      setCustomImage(json.images?.[0] || '');
   161→      toast.success('Haber içeriği getirildi');
   162→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
   163→    finally { setFetching(false); }
   164→  };
   165→
   166→  const handleSaveCustom = async () => {
   167→    if (!customTitle.trim() || !customSummary.trim()) return;
   168→    setSaving(true);
   169→    try {
   170→      const r = await fetch('/api/admin/custom-article', { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` }, body: JSON.stringify({ action: 'save', title: customTitle, summary: customSummary, imageUrl: customImage || null, category: customCategory }) });
   171→      const json = (await r.json()) as { ok?: boolean; error?: string };
   172→      if (!r.ok || !json.ok) throw new Error(json.error || 'Kaydedilemedi');
   173→      toast.success('Özel haber yayınlandı');
   174→      setFetchUrl(''); setCustomTitle(''); setCustomSummary(''); setCustomImage(''); setFetchedImages([]);
   175→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
   176→    finally { setSaving(false); }
   177→  };
   178→
   179→  // Published article handlers
   180→  const startEdit = (a: PubArticle) => { setEditingId(a.id); setEditTitle(a.aiTitle); setEditSummary(a.aiSummary); setEditImage(a.imageUrl || ''); };
   181→  const cancelEdit = () => { setEditingId(null); setEditTitle(''); setEditSummary(''); setEditImage(''); };
   182→  const saveEdit = async (id: string) => {
   183→    setSavingEdit(true);
   184→    try {
   185→      const r = await fetch(`/api/admin/published/${id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` }, body: JSON.stringify({ aiTitle: editTitle, aiSummary: editSummary, imageUrl: editImage || null }) });
   186→      if (!r.ok) throw new Error('Güncellenemedi');
   187→      setPubArticles(a => a.map(x => x.id === id ? { ...x, aiTitle: editTitle, aiSummary: editSummary, imageUrl: editImage || null } : x));
   188→      toast.success('Haber güncellendi'); cancelEdit();
   189→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
   190→    finally { setSavingEdit(false); }
   191→  };
   192→  const deletePub = async (id: string) => {
   193→    try {
   194→      const r = await fetch(`/api/admin/published/${id}`, { method: 'DELETE', headers: { Authorization: `Bearer ${token}` } });
   195→      if (!r.ok) throw new Error('Silinemedi');
   196→      setPubArticles(a => a.filter(x => x.id !== id)); toast.success('Haber silindi');
   197→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Hata'); }
   198→  };
   199→
   200→  // Reset handlers
   201→  const handleReset = async () => {
   202→    setResetting(true); setResetResult(null);
   203→    try {
   204→      const r = await fetch('/api/admin/reset', { method: 'POST', headers: { Authorization: `Bearer ${token}` } });
   205→      const json = (await r.json()) as { ok?: boolean; deleted?: { publishedArticles: number; articles: number; readerMessages: number }; error?: string };
   206→      if (!r.ok || !json.ok) throw new Error(json.error || 'Sıfırlama başarısız');
   207→      const d = json.deleted!;
   208→      setResetResult(`Silindi: ${d.articles} makale, ${d.publishedArticles} özet, ${d.readerMessages} mesaj. Yenileme başlatıldı.`);
   209→      setMessages([]); setSelectedIds(new Set()); setPubArticles([]);
   210→      setTimeout(() => window.location.reload(), 3000);
   211→    } catch (e) { setResetResult(e instanceof Error ? e.message : 'Hata'); }
   212→    finally { setResetting(false); }
   213→  };
   214→
   215→  const [expandedSources, setExpandedSources] = useState<string | null>(null);
   216→  const [sourceLinks, setSourceLinks] = useState<{ [key: string]: Array<{ title: string; link: string; source: string }> }>({});
   217→  const [loadingSources, setLoadingSources] = useState(false);
   218→
   219→  const toggleSources = async (articleId: string, sourceIdsJson: string) => {
   220→    if (expandedSources === articleId) {
   221→      setExpandedSources(null);
   222→      return;
   223→    }
   224→    setExpandedSources(articleId);
   225→    if (sourceLinks[articleId]) return;
   226→
   227→    setLoadingSources(true);
   228→    try {
   229→      let ids: string[] = [];
   230→      try { ids = JSON.parse(sourceIdsJson); } catch { /* ignore */ }
   231→      if (ids[0] === 'custom' || ids.length === 0) {
   232→        setSourceLinks(prev => ({ ...prev, [articleId]: [] }));
   233→        return;
   234→      }
   235→      const results = await Promise.all(ids.map(async (id) => {
   236→        try {
   237→          const r = await fetch(`/api/articles/${id}`, { cache: 'no-store' });
   238→          if (!r.ok) return null;
   239→          const json = (await r.json()) as { article: { title: string; link: string; source: { name: string } } };
   240→          return { title: json.article.title, link: json.article.link, source: json.article.source.name };
   241→        } catch { return null; }
   242→      }));
   243→      const valid = results.filter((r): r is { title: string; link: string; source: string } => r !== null);
   244→      setSourceLinks(prev => ({ ...prev, [articleId]: valid }));
   245→    } catch { /* ignore */ }
   246→    finally { setLoadingSources(false); }
   247→  };
   248→
   249→  const handleLogout = () => { localStorage.removeItem('admin_token'); setToken(null); setMessages([]); setSelectedIds(new Set()); setPubArticles([]); setAdminTab('messages'); };
   250→  const newCount = messages.filter(m => m.status === 'new').length;
   251→  const allSelected = messages.length > 0 && selectedIds.size === messages.length;
   252→  const resetConfirmed = resetConfirm.trim().toLowerCase() === 'evet';
   253→
   254→  // File upload handler (for both custom article and published edit)
   255→  const [uploading, setUploading] = useState(false);
   256→  const handleFileUpload = async (file: File, onDone: (url: string) => void) => {
   257→    setUploading(true);
   258→    const formData = new FormData();
   259→    formData.append('file', file);
   260→    try {
   261→      const r = await fetch('/api/admin/upload', { method: 'POST', headers: { Authorization: `Bearer ${token}` }, body: formData });
   262→      const json = (await r.json()) as { ok?: boolean; url?: string; error?: string };
   263→      if (!r.ok || !json.ok) throw new Error(json.error || 'Yükleme hatası');
   264→      onDone(json.url!);
   265→      toast.success('Görsel yüklendi');
   266→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Yükleme hatası'); }
   267→    finally { setUploading(false); }
   268→  };
   269→
   270→  return (
   271→    <>
   272→      <Dialog open={open} onOpenChange={(o) => { if (!o) onClose(); }}>
   273→        <DialogContent className="max-w-3xl p-0">
   274→          <DialogHeader className="px-6 pt-6 pb-0">
   275→            <DialogTitle className="flex items-center justify-between">
   276→              <span className="flex items-center gap-2"><Lock className="h-5 w-5" /> Yönetici Paneli</span>
   277→              {token && (
   278→                <div className="flex items-center gap-2">
   279→                  <Button variant="outline" size="sm" onClick={() => setResetOpen(true)} className="gap-1.5 text-xs text-destructive hover:text-destructive border-destructive/30">
   280→                    <RotateCcw className="h-3.5 w-3.5" /> Siteyi Sıfırla
   281→                  </Button>
   282→                  <Button variant="ghost" size="sm" onClick={handleLogout} className="text-xs">Çıkış</Button>
   283→                </div>
   284→              )}
   285→            </DialogTitle>
   286→          </DialogHeader>
   287→
   288→          <div className="max-h-[72vh] overflow-y-auto news-scroll px-6 pb-6 pt-2">
   289→            {!token ? (
   290→              <form onSubmit={handleLogin} className="mx-auto max-w-sm space-y-4 py-8">
   291→                <div className="text-center">
   292→                  <div className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-full bg-muted"><Lock className="h-6 w-6 text-muted-foreground" /></div>
   293→                  <h2 className="text-lg font-bold">Abone Girişi</h2>
   294→                  <p className="mt-1 text-sm text-muted-foreground">Telefonunuza gelen kodu giriniz</p>
   295→                </div>
   296→                <div className="space-y-2">
   297→                  <label className="text-xs text-muted-foreground">Telefon numaranızı giriniz</label>
   298→                  <Input type="tel" placeholder="05XX XXX XX XX" className="text-center" />
   299→                </div>
   300→                <div className="space-y-2">
   301→                  <label className="text-xs text-muted-foreground">Kodu giriniz</label>
   302→                  <Input type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="• • • • • •" autoFocus className="text-center" />
   303→                </div>
   304→                <Button type="submit" disabled={loggingIn || !password.trim()} className="w-full gap-2">{loggingIn ? <Loader2 className="h-4 w-4 animate-spin" /> : <Lock className="h-4 w-4" />} Giriş Yap</Button>
   305→              </form>
   306→            ) : (
   307→              <>
   308→                {/* Sub-tabs */}
   309→                <div className="mb-4 flex gap-1 rounded-lg border border-border bg-muted/30 p-1">
   310→                  {([['messages', 'Mesajlar', Mail], ['custom', 'Özel Haber Ekle', Star], ['published', 'Yayındaki Haberler', Newspaper]] as const).map(([id, label, Icon]) => (
   311→                    <button key={id} type="button" onClick={() => setAdminTab(id)} className={`flex flex-1 items-center justify-center gap-1.5 rounded-md px-3 py-2 text-xs font-medium transition ${adminTab === id ? 'bg-secondary text-secondary-foreground shadow-sm' : 'text-muted-foreground hover:text-foreground'}`}>
   312→                      <Icon className="h-3.5 w-3.5" /> {label}
   313→                    </button>
   314→                  ))}
   315→                </div>
   316→
   317→                {/* Messages tab */}
   318→                {adminTab === 'messages' && (
   319→                  <div>
   320→                    {messages.length > 0 && (
   321→                      <div className="mb-3 flex flex-wrap items-center gap-2">
   322→                        <Button variant="ghost" size="sm" onClick={selectAll} disabled={bulkDeleting} className="gap-1.5 text-xs">{allSelected ? <CheckSquare className="h-3.5 w-3.5" /> : <Square className="h-3.5 w-3.5" />}{allSelected ? 'Seçimi Kaldır' : 'Tümünü Seç'}</Button>
   323→                        <Button variant="outline" size="sm" onClick={deleteSelected} disabled={bulkDeleting || selectedIds.size === 0} className="gap-1.5 text-xs">{bulkDeleting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Trash2 className="h-3.5 w-3.5" />}Seçilenleri Sil ({selectedIds.size})</Button>
   324→                        <AlertDialog><AlertDialogTrigger asChild><Button variant="outline" size="sm" disabled={bulkDeleting} className="gap-1.5 text-xs text-destructive hover:text-destructive"><CheckCheck className="h-3.5 w-3.5" />Tümünü Sil</Button></AlertDialogTrigger>
   325→                          <AlertDialogContent><AlertDialogHeader><AlertDialogTitle>Tüm mesajları sil?</AlertDialogTitle><AlertDialogDescription>{messages.length} mesajın tamamı silinecek.</AlertDialogDescription></AlertDialogHeader>
   326→                          <AlertDialogFooter><AlertDialogCancel>Vazgeç</AlertDialogCancel><AlertDialogAction onClick={deleteAllMsgs} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">Tümünü Sil</AlertDialogAction></AlertDialogFooter></AlertDialogContent>
   327→                        </AlertDialog>
   328→                      </div>
   329→                    )}
   330→                    {loadingMsgs ? <div className="space-y-3">{[1,2,3].map(i => <Skeleton key={i} className="h-24 w-full rounded-xl" />)}</div>
   331→                    : messages.length === 0 ? <Card className="flex flex-col items-center gap-3 p-10 text-center"><Mail className="h-10 w-10 text-muted-foreground" /><p className="text-sm text-muted-foreground">Henüz okuyucu mesajı yok</p></Card>
   332→                    : <div className="space-y-3">{messages.map(m => { const sel = selectedIds.has(m.id); return (
   333→                      <Card key={m.id} className={`p-4 ${m.status === 'new' ? 'border-news/40 bg-news/[0.04]' : ''} ${sel ? 'ring-2 ring-news/40' : ''}`}>
   334→                        <div className="flex items-start gap-3"><Checkbox checked={sel} onCheckedChange={() => toggleSelect(m.id)} className="mt-1" />
   335→                          <div className="min-w-0 flex-1"><div className="flex items-center gap-2"><h4 className="text-sm font-semibold">{m.name}</h4>{m.status === 'new' && <Badge className="bg-news text-news-foreground text-[9px]">YENİ</Badge>}<span className="ml-auto inline-flex items-center gap-1 text-[10px] text-muted-foreground"><Clock className="h-3 w-3" />{new Date(m.createdAt).toLocaleString('tr-TR')}</span></div>
   336→                            <p className="mt-0.5 text-xs text-muted-foreground">{m.email} · {m.subject}</p><p className="mt-2 text-sm leading-relaxed text-foreground/80">{m.message}</p></div>
   337→                          <Button variant="ghost" size="icon" onClick={() => handleDelete(m.id)} disabled={deletingId === m.id} className="h-8 w-8 flex-shrink-0 text-muted-foreground hover:text-destructive">{deletingId === m.id ? <Loader2 className="h-4 w-4 animate-spin" /> : <Trash2 className="h-4 w-4" />}</Button>
   338→                        </div></Card>); })}</div>}
   339→                  </div>
   340→                )}
   341→
   342→                {/* Custom article tab */}
   343→                {adminTab === 'custom' && (
   344→                  <div className="space-y-4">
   345→                    <form onSubmit={handleFetch} className="flex gap-2">
   346→                      <Input value={fetchUrl} onChange={(e) => setFetchUrl(e.target.value)} placeholder="https://ornek.com/haber-basligi" className="flex-1" />
   347→                      <Button type="submit" disabled={fetching || !fetchUrl.trim()} className="gap-2">{fetching ? <Loader2 className="h-4 w-4 animate-spin" /> : <Globe className="h-4 w-4" />}Getir</Button>
   348→                    </form>
   349→                    {customTitle && (
   350→                      <div className="space-y-4 rounded-lg border border-border p-4">
