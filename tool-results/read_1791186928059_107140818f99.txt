    56→globalThis.fetch = function(url, options) {
    57→  options = options || {};
    58→  return new Promise(function(resolve, reject) {
    59→    var lib = url.indexOf('https') === 0 ? require('https') : require('http');
    60→    var urlObj = new URL(url);
    61→    var headers = Object.assign({}, options.headers || {});
    62→    if (options.body) headers['Content-Length'] = Buffer.byteLength(options.body);
    63→    var req = lib.request({
    64→      hostname: urlObj.hostname,
    65→      port: urlObj.port || (urlObj.protocol === 'https:' ? 443 : 80),
    66→      path: urlObj.pathname + urlObj.search,
    67→      method: options.method || 'GET',
    68→      headers: headers
    69→    }, function(resp) {
    70→      var chunks = [];
    71→      resp.on('data', function(c) { chunks.push(c); });
    72→      resp.on('end', function() {
    73→        var text = Buffer.concat(chunks).toString('utf8');
    74→        resolve({ status: resp.statusCode, ok: resp.statusCode >= 200 && resp.statusCode < 300, json: function() { return Promise.resolve(JSON.parse(text)); }, text: function() { return Promise.resolve(text); } });
    75→      });
    76→    });
    77→    req.on('error', reject);
    78→    req.setTimeout(120000, function() { req.destroy(); reject(new Error('timeout')); });
    79→    if (options.body) req.write(options.body);
    80→    req.end();
    81→  });
    82→};

   197→function normalizeTitle(t) {
   198→  if (!t) return '';
   199→  // HTML entity decode (&#039; -> ', &amp; -> &, &quot; -> ", &nbsp; -> space, &#x27; -> ')
   200→  return String(t)
   201→    .replace(/&#x([0-9a-fA-F]+);/g, function(_, h) { return String.fromCharCode(parseInt(h, 16)); })
   202→    .replace(/&#(\d+);/g, function(_, d) { return String.fromCharCode(parseInt(d, 10)); })
   203→    .replace(/&amp;/g, '&')
   204→    .replace(/&quot;/g, '"')
   205→    .replace(/&apos;/g, "'")
   206→    .replace(/&nbsp;/g, ' ')
   207→    .replace(/&lt;/g, '<')
   208→    .replace(/&gt;/g, '>')
   209→    .toLowerCase()
   210→    .replace(/[''`]/g, "'")
   211→    .replace(/[^\w\sçğıöşü]/g, ' ')
   212→    .replace(/\s+/g, ' ')
   213→    .trim();
   214→}

   224→function titleSimilar(t1, t2) {
   225→  var n1 = normalizeTitle(t1), n2 = normalizeTitle(t2);
   226→  if (!n1 || !n2) return 0;
   227→  // Birebir aynı başlık → kesin eşleşme (%100)
   228→  if (n1 === n2) return 1.0;
   229→  // 4+ karakter ve stop word olmayan kelimeleri al
   230→  var w1 = n1.split(' ').filter(function(w) { return w.length > 3 && !STOP_WORDS.has(w); });
   231→  var w2 = n2.split(' ').filter(function(w) { return w.length > 3 && !STOP_WORDS.has(w); });
   232→  if (!w1.length || !w2.length) return 0;
   233→  var set2 = new Set(w2); var common = 0;
   234→  w1.forEach(function(w) { if (set2.has(w)) common++; });
   235→  var ratio = common / Math.max(w1.length, w2.length);
   236→  // Sadece 1 ortak kelime varsa benzerlik yetersiz (en az 2 anlamlı kelime şart)
   237→  if (common < 2) return 0;
   238→  return ratio;
   239→}

   242→function contentSimilar(c1, c2) {
   243→  if (!c1 || !c2) return 0;
   244→  var n1 = normalizeTitle(c1.slice(0, 1500));
   245→  var n2 = normalizeTitle(c2.slice(0, 1500));
   246→  var w1 = n1.split(' ').filter(function(w) { return w.length > 4 && !STOP_WORDS.has(w); });
   247→  var w2 = n2.split(' ').filter(function(w) { return w.length > 4 && !STOP_WORDS.has(w); });
   248→  if (!w1.length || !w2.length) return 0;
   249→  var set2 = new Set(w2); var common = 0;
   250→  w1.forEach(function(w) { if (set2.has(w)) common++; });
   251→  return common / Math.max(w1.length, w2.length);
   252→}

   254→// Gruplama — başlık benzerliği 0.3 + içerik benzerliği 0.4 (combo)
   255→// Eşik yüksek — alakasız haberler aynı grupta birleşmesin
   256→// "İstanbul baskını" vs "İstanbul gastronomi" -> titleSimilar düşük, ayrı grup
   257→function groupArticles(articles) {
   258→  var groups = [], used = new Set();
   259→  for (var i = 0; i < articles.length; i++) {
   260→    if (used.has(i)) continue;
   261→    var group = { articles: [articles[i]], sourceIds: new Set([articles[i].sourceId]) };
   262→    used.add(i);
   263→    for (var j = i + 1; j < articles.length; j++) {
   264→      if (used.has(j)) continue;
   265→      // Kategori BAĞIMSIZ — aynı haber farklı kategori etiketiyle gelirse yine eşleşir
   266→      var titleSim = titleSimilar(articles[i].title, articles[j].title);
   267→      var contentSim = contentSimilar(articles[i].content || articles[i].description || '', articles[j].content || articles[j].description || '');
   268→      // Eşik: title >= 0.3 (yüksek) VEYA (title >= 0.15 + content >= 0.4 yüksek)
   269→      // Bu, alakasız haberlerin aynı grupta olmasını önler
   270→      if (titleSim >= 0.3 || (titleSim >= 0.15 && contentSim >= 0.4)) {
   271→        group.articles.push(articles[j]);
   272→        group.sourceIds.add(articles[j].sourceId);
   273→        used.add(j);
   274→      }
   275→    }
   276→    groups.push(group);
   277→  }
   278→  return groups;
   279→}

   418→        // Önceki: firstArticle.category (rastgele) → yanlış kategori
   419→        var catSourcesByCat = {};
   420→        for (var gi = 0; gi < groupArticlesList.length; gi++) {
   421→          var ga = groupArticlesList[gi];
   422→          var gcat = ga.category || 'Siyaset';
   423→          if (!catSourcesByCat[gcat]) catSourcesByCat[gcat] = new Set();
   424→          catSourcesByCat[gcat].add(ga.sourceId);
   425→        }
   426→        var bestCat = firstArticle.category || 'Siyaset';
   427→        var bestCount = 0;
   428→        Object.keys(catSourcesByCat).forEach(function(c) {
   429→          if (catSourcesByCat[c].size > bestCount) {
   430→            bestCount = catSourcesByCat[c].size;
   431→            bestCat = c;
   432→          }
   433→        });
   434→        var cat = bestCat;
   435→        var sourceCount = group.sourceIds.size;
   436→
   437→        // TEKRAR KONTROLÜ — kategori BAĞIMSIZ, eşik DÜŞÜK (0.3)
   438→        // Yeni mantık (çok sıkı tekrar yakalama):
   439→        // - titleSimilar >= 0.85 (birebir) + published ise: SKIP (create etme, AI çağırma)
   440→        //   existing'i güncelle (latestPublishedAt, sourceCount)
   441→        // - titleSimilar >= 0.85 + archived/duplicate ise: yeni published create
   442→        // - titleSimilar 0.3-0.85: eski duplicate, yeni published (AI yeni)
   443→        // - titleSimilar 0.15-0.3 + contentSimilar >= 0.3: eski duplicate
   444→        // - < 0.15: published (yeni haber)
   445→        // Tüm yeni haberler direkt PUBLISHED — pending_review YOK
   446→        var oldArticleToDuplicate = null;
   447→        var alreadyArchived = false;
   448→        var skipCreate = false;  // birebir aynı başlık + published → create etme
   449→
   450→        // Önce benzer başlık ara — kategori fark etmez
   451→        for (var j = 0; j < existing.length; j++) {
   452→          var simTitle = titleSimilar(firstArticle.title, existing[j].aiTitle || '');
   453→          var simContent = contentSimilar(
   454→            firstArticle.content || firstArticle.description || '',
   455→            existing[j].aiSummary || ''
   456→          );
   457→          // Eşik: title >= 0.3 VEYA (title >= 0.15 VE content >= 0.3)
   458→          var matched = simTitle >= 0.3 || (simTitle >= 0.15 && simContent >= 0.3);
   459→          if (!matched) continue;
   460→
   461→          if (simTitle >= 0.85 && existing[j].status === 'published') {
   462→            // Birebir aynı başlık + published → SKIP (create etme, AI çağırma)
   463→            // existing'i güncelle (latestPublishedAt, sourceCount)
   464→            skipCreate = true;
   465→            try {
   466→              await globalThis.prisma.publishedArticle.update({
   467→                where: { id: existing[j].id },
   468→                data: {
   469→                  latestPublishedAt: new Date(),
   470→                  sourceCount: Math.max(existing[j].sourceCount || 0, sourceCount)
   471→                }
   472→              });
   473→              log('  [Birebir ayni, skip + update] (' + cat + '): ' + firstArticle.title.slice(0, 50));
   474→            } catch (e) {}
   475→            break;
   476→          }
   477→
   478→          if (existing[j].status === 'published') {
   479→            oldArticleToDuplicate = existing[j];
   480→            log('  [Tekrar bulundu, eski -> duplicate] (' + cat + '/' + (existing[j].category || '?') + '): ' + firstArticle.title.slice(0, 50) + ' (benzerlik: ' + Math.round(simTitle * 100) + '%)');
   481→            break;
   482→          } else if (existing[j].status === 'archived' || existing[j].status === 'duplicate') {
   483→            alreadyArchived = true;
   484→            break;
   485→          }
   486→        }
   487→
   488→        if (skipCreate) {
   489→          skipped++;
   490→          continue;  // AI çağırma, create etme, sonraki gruba geç
   491→        }
   492→
   493→        if (alreadyArchived) {
   494→          log('  [Arsivde/Duplicatete var, taze yayinla] (' + cat + '): ' + firstArticle.title.slice(0, 50));
   495→        }
   496→        if (oldArticleToDuplicate) {
   497→          try {
   498→            await globalThis.prisma.publishedArticle.update({
   499→              where: { id: oldArticleToDuplicate.id },
   500→              data: { status: 'duplicate', archivedAt: new Date() }
   501→            });
   502→            archived++;
   503→            log('  [Eski -> duplicate] (' + cat + '): ' + (oldArticleToDuplicate.aiTitle || '').slice(0, 50));
   504→          } catch (e) {}
   505→        }
   506→
   507→        log('  [' + (i+1) + '/' + topGroups.length + '] ' + sourceCount + ' kaynak, ' + groupArticlesList.length + ' makale (' + cat + '): ' + firstArticle.title.slice(0, 50));
   508→
   509→        // İçerikleri topla — Siyaset/Ekonomi: en yeni 5 kaynak, diğerleri: en yeni 2
   510→        var maxSources = (cat === 'Siyaset' || cat === 'Ekonomi / Finans') ? 5 : 2;
   511→        var contents = groupArticlesList.slice(0, maxSources).map(function(a) { return a.content || a.description || ''; });
   512→
   513→        // AI özet
   514→        var aiText = await aiSummarize(firstArticle.title, contents, cat);
   515→        var summaryText = aiText || (firstArticle.description ? firstArticle.description.slice(0, 500) : firstArticle.title || '');
   516→        if (aiText) aiOk++;
   517→
   518→        // En iyi görsel
   519→        var bestImage = null;
   520→        for (var k = 0; k < groupArticlesList.length; k++) {
   521→          if (groupArticlesList[k].imageUrl) { bestImage = groupArticlesList[k].imageUrl; break; }
   522→        }
   523→
   524→        try {
   525→          var publishTime = new Date(Date.now() - i * 60000);
   526→          // Tüm yeni haberler direkt PUBLISHED — pending_review KALDIRILDI
   527→          var createdArticle = await globalThis.prisma.publishedArticle.create({
   528→            data: {
   529→              aiTitle: decodeHtmlEntities(firstArticle.title),
   530→              aiSummary: summaryText.slice(0, 2000),
   531→              category: cat,
   532→              imageUrl: bestImage,
   533→              sourceArticleIds: JSON.stringify(groupArticlesList.map(function(a) { return a.id; })),
   534→              sourceCount: sourceCount,
   535→              earliestPublishedAt: groupArticlesList[groupArticlesList.length - 1].publishedAt || publishTime,
   536→              latestPublishedAt: publishTime,
   537→              wordCount: summaryText.split(/\s+/).length,
   538→              status: 'published',
   539→              publishedAt: publishTime
   540→            }
   541→          });
   542→          added++;
   543→          // CRITICAL: yeni create edileni existing listesine ekle
   544→          // ki cycle içinde aynı başlıkla başka grup gelirse onu görelim
   545→          existing.push({
   546→            id: createdArticle.id,
   547→            aiTitle: decodeHtmlEntities(firstArticle.title),
   548→            aiSummary: summaryText.slice(0, 2000),
   549→            category: cat,
   550→            status: 'published'
   551→          });
   552→        } catch (e) { log('  DB hata: ' + e.message); }
   553→      }
   554→      log('Added: ' + added + ', Archived: ' + archived + ', AI: ' + aiOk);
   555→
   556→      // Yayın süresi dolan haberleri arşive taşı (6 saatten eski published)
   557→      // Bu sayede site "taze" kalır — aynı başlıklar 6 saat sonra kalkar, yenileri gelir
   558→      try {
   559→        var staleCutoff = new Date(Date.now() - 6 * 60 * 60 * 1000); // 6 saat
   560→        var staleResult = await globalThis.prisma.publishedArticle.updateMany({
   561→          where: {
   562→            status: 'published',
   563→            latestPublishedAt: { lt: staleCutoff }
   564→          },
   565→          data: { status: 'archived', archivedAt: new Date() }
   566→        });
   567→        if (staleResult.count > 0) {
   568→          log('Yayın süresi dolan haber arşive taşındı: ' + staleResult.count + ' (6 saatten eski)');
   569→        }
   570→      } catch (e) { log('Stale arşiv hatası: ' + e.message); }
   571→
   572→      // ====== POST-CYCLE CLEANUP ======
   573→      // Pipeline bittikten sonra tüm published'ları tara,
   574→      // aynı/benzer başlığa sahip eskileri SİL (duplicate'e atma, direkt sil)
   575→      // Bu sayede "aynı haber farklı saatlerde" durumu olmaz
   576→      try {
   577→        var allPublished = await globalThis.prisma.publishedArticle.findMany({
   578→          where: { status: 'published' },
   579→          select: { id: true, aiTitle: true, aiSummary: true, category: true, latestPublishedAt: true },
   580→          orderBy: { latestPublishedAt: 'desc' }  // en yeni ilk
   581→        });
   582→
   583→        var seenTitles = [];  // [{ title, summary, id }]
   584→        var toDelete = [];
   585→
   586→        for (var pi = 0; pi < allPublished.length; pi++) {
   587→          var pub = allPublished[pi];
   588→          var isDuplicate = false;
   589→
   590→          for (var si = 0; si < seenTitles.length; si++) {
   591→            var sim = titleSimilar(pub.aiTitle, seenTitles[si].aiTitle || '');
   592→            var simC = contentSimilar(pub.aiSummary || '', seenTitles[si].aiSummary || '');
   593→            // Eşik: title >= 0.3 veya (title >= 0.15 + content >= 0.3)
   594→            if (sim >= 0.3 || (sim >= 0.15 && simC >= 0.3)) {
   595→              // Bu pub daha eski (çünkü allPublished latestPublishedAt desc sıralı)
   596→              // Yani pub'ı sil
   597→              isDuplicate = true;
   598→              break;
   599→            }
   600→          }
   601→
   602→          if (isDuplicate) {
   603→            toDelete.push(pub.id);
   604→          } else {
   605→            seenTitles.push(pub);
   606→          }
   607→        }
   608→
   609→        if (toDelete.length > 0) {
   610→          var delResult = await globalThis.prisma.publishedArticle.deleteMany({
   611→            where: { id: { in: toDelete } }
   612→          });
   613→          log('Post-cycle cleanup: ' + delResult.count + ' tekrar haber silindi (ana sayfada tekrar kalmadi)');
   614→        }
   615→      } catch (e) { log('Post-cleanup hatasi: ' + e.message); }
   616→
   617→      // Eski duplicate'leri sil (3 günden eski) — DB şişmesin
   618→      try {
   619→        var dupCutoff = new Date(Date.now() - 3 * 24 * 60 * 60 * 1000);
   620→        var oldDups = await globalThis.prisma.publishedArticle.deleteMany({
   621→          where: { status: 'duplicate', archivedAt: { lt: dupCutoff } }
   622→        });
   623→        if (oldDups.count > 0) {
   624→          log('Eski duplicate silindi: ' + oldDups.count + ' kayit (3 günden eski)');
   625→        }
   626→      } catch (e) { log('Eski duplicate silme hatasi: ' + e.message); }
   627→
   628→      // Eski archived haberleri sil (15 günden eski) — DB şişmesin
   629→      try {
   630→        var archCutoff = new Date(Date.now() - 15 * 24 * 60 * 60 * 1000);
   631→        var oldArchived = await globalThis.prisma.publishedArticle.deleteMany({
   632→          where: { status: 'archived', archivedAt: { lt: archCutoff } }
   633→        });
   634→        if (oldArchived.count > 0) {
   635→          log('Eski archived silindi: ' + oldArchived.count + ' kayit (15 günden eski)');
   636→        }
   637→      } catch (e) { log('Eski archived silme hatasi: ' + e.message); }
   638→
   639→      // Arşiv: kategori bazlı 15'den fazla varsa eskiyi arşive taşı
   640→      for (var ci = 0; ci < Object.keys(CATEGORY_MIN_SOURCES).length; ci++) {
   641→        var categoryName = Object.keys(CATEGORY_MIN_SOURCES)[ci];
   642→        var catArticles = await globalThis.prisma.publishedArticle.findMany({
   643→          where: { status: 'published', category: categoryName },
   644→          orderBy: { latestPublishedAt: 'desc' },
   645→          select: { id: true }
   646→        });
   647→        if (catArticles.length > CATEGORY_MAX) {
   648→          var toArchive = catArticles.slice(CATEGORY_MAX);
   649→          for (var ai = 0; ai < toArchive.length; ai++) {
   650→            await globalThis.prisma.publishedArticle.update({
   651→              where: { id: toArchive[ai].id },
   652→              data: { status: 'archived', archivedAt: new Date() }
   653→            });
   654→          }
   655→          log('Arşive taşındı: ' + categoryName + ' ' + toArchive.length + ' haber');
   656→        }
   657→      }
   658→
   659→      // Ana sayfa: toplam 50'den fazla published varsa eskiyi arşive taşı
   660→      var totalPublished = await globalThis.prisma.publishedArticle.count({ where: { status: 'published' } });
   661→      if (totalPublished > HOME_MAX_TOTAL) {
   662→        var allPublished = await globalThis.prisma.publishedArticle.findMany({
   663→          where: { status: 'published' },
   664→          orderBy: { latestPublishedAt: 'desc' },
   665→          select: { id: true }
   666→        });
   667→        var toArchiveTotal = allPublished.slice(HOME_MAX_TOTAL);
   668→        for (var ti = 0; ti < toArchiveTotal.length; ti++) {
   669→          await globalThis.prisma.publishedArticle.update({
   670→            where: { id: toArchiveTotal[ti].id },
   671→            data: { status: 'archived', archivedAt: new Date() }
   672→          });
   673→        }
   674→        log('Ana sayfa arşive taşındı: ' + toArchiveTotal.length + ' haber');
   675→      }
   676→
   677→      ws({ summariesDone: aiOk });
   678→    } catch (e) {
   679→      log('Publish hatasi: ' + e.message);
   680→    }
   681→  }
   682→
   683→  // Hatalı RSS kaynaklarını deaktif et (10 hata = pasif)
   684→  try {
   685→    var failedSources = await globalThis.prisma.source.findMany({
   686→      where: { active: true },
   687→      select: { id: true, name: true, url: true }
   688→    });
   689→    // Bu adımı trigger-refresh.js içinde hata sayımı yapacak şekilde güncelleyeceğiz
   690→    // Şimdilik sadece log
   691→    log('Aktif kaynak: ' + failedSources.length);
   692→  } catch (e) {}
   693→
   694→  ws({ stage: 'done', finishedAt: new Date().toISOString() });
   695→  log('=== Cycle tamam ===');
   696→  process.exit = origExit;
   697→  origExit(0);
   698→}
